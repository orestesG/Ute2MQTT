#include <tunables/global>

profile hassio_addon_ute2mqtt flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/python>
  #include <abstractions/ssl_certs>

  # Signals
  signal (send, receive),

  # Network: MQTT (1883) + HTTPS to UTE portal (443) + DNS (53/udp)
  network inet tcp,
  network inet6 tcp,
  network inet udp,
  network inet6 udp,

  # Entrypoint
  /run.sh ix,

  # Shell tools used in run.sh
  /bin/sh ix,
  /bin/bash ix,
  /bin/dash ix,
  /usr/bin/env ix,
  /usr/bin/cat ix,
  /bin/rm ix,
  /usr/bin/rm ix,

  # Python interpreter
  /usr/bin/python3* ix,
  /usr/local/bin/python3* ix,

  # Application code (read-only)
  /app/ r,
  /app/** r,

  # Data: credentials, encryption_key, options.json
  /data/ r,
  /data/** rw,

  # Temp files (Python internals)
  /tmp/ r,
  /tmp/** rw,

  # Proc (minimal, needed by Python)
  @{PROC}/self/ r,
  @{PROC}/self/fd/ r,
  @{PROC}/self/fd/** rw,
  @{PROC}/self/maps r,

  # S6-Overlay init system (imagen base de HA)
  /init ix,
  /run/s6/ r,
  /run/s6/** rwix,
  /run/s6-rc*/ r,
  /run/s6-rc*/** rwix,
  /etc/s6-overlay/ r,
  /etc/s6-overlay/** r,
  /run/fix-attrs.d/ r,
  /run/fix-attrs.d/** r,

  # Bashio (helper shell de HA)
  /usr/lib/bashio/ r,
  /usr/lib/bashio/** ix,

  # Binarios del sistema (invocados por S6 internamente)
  /bin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /sbin/** ix,

  # Capabilities mínimas para S6
  capability chown,
  capability setuid,
  capability setgid,
  capability fowner,
  capability fsetid,
  capability dac_override,
}
